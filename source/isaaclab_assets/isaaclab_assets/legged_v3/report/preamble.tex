% ===== NGÔN NGỮ VIỆT & FONT =====
% Tương thích cả pdfLaTeX lẫn XeLaTeX
\usepackage{iftex}
\ifXeTeX
    \usepackage{fontspec}
    \setmainfont{Times New Roman}
\else
    \usepackage[utf8]{inputenc}
    \usepackage[T5]{fontenc}        % Encoding tiếng Việt cho pdfLaTeX
    \usepackage{mathptmx}           % Times New Roman clone cho pdfLaTeX
\fi
\usepackage[vietnamese]{babel}      % Tiếng Việt: caption, hyphenation

% Ghi đè caption tiếng Việt (babel overrides \renewcommand nên phải dùng \addto)
\addto\captionsvietnamese{%
    \renewcommand{\contentsname}{MỤC LỤC}%
    \renewcommand{\listfigurename}{DANH MỤC HÌNH ẢNH}%
    \renewcommand{\listtablename}{DANH MỤC BẢNG}%
    \renewcommand{\chaptername}{CHƯƠNG}%
}

% ===== KHỔ GIẤY & LỀ =====
\usepackage[a4paper,
    top=20mm,
    bottom=20mm,
    left=30mm,
    right=20mm]{geometry}           % Lề: trái 3cm, trên/dưới/phải 2cm

% ===== GÓI CẦN THIẾT =====
\usepackage{graphicx}
\usepackage[dvipsnames]{xcolor}
\usepackage{amsmath}
\usepackage{fancyhdr}
\usepackage{setspace}
\usepackage{changepage}
\usepackage{etoolbox}
\usepackage{titlesec}
\usepackage{tocloft}
\usepackage{tikz}
\usetikzlibrary{calc}
\usepackage{enumitem}
\usepackage{siunitx}
\usepackage{csquotes}
\usepackage{float}
\usepackage{caption}
\usepackage{adjustbox}
\usepackage{url}
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{multirow}

% hyperref load sau cùng — tắt màu xanh, dùng hidelinks
\usepackage[colorlinks=true,
    linkcolor=black,
    urlcolor=black,
    citecolor=black]{hyperref}

% ===== LISTINGS (hiển thị mã nguồn) =====
\usepackage{listings}
\lstset{
    basicstyle=\ttfamily\small,
    breaklines=true,
    backgroundcolor=\color{gray!10},
    keywordstyle=\color{blue},
    commentstyle=\color{gray},
    stringstyle=\color{red},
    numbers=left,
    numberstyle=\tiny\color{gray},
    frame=single
}
\lstnewenvironment{pythoncode}{\lstset{language=Python}}{}

% ===== CẤU HÌNH TOÀN CỤC =====

% Header/footer: số trang giữa footer
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}

% Chỉ hiện MỤC LỤC đến cấp section (1.1), không hiện subsection (1.1.1)
\setcounter{tocdepth}{1}

% Paragraph
\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt}

% Giãn dòng 1.5 theo yêu cầu
\setstretch{1.5}

% Đảm bảo body text luôn là 13pt (tránh extreport fallback)
\AtBeginDocument{\fontsize{13pt}{15.6pt}\selectfont}

% ===== CẤU HÌNH MỤC LỤC (tocloft) =====
% Tiêu đề MỤC LỤC / DANH MỤC: căn giữa, 16pt, đậm
\renewcommand{\cfttoctitlefont}{\hfil\fontsize{16pt}{18pt}\selectfont\bfseries}
\renewcommand{\cftaftertoctitle}{\hfil}
\setlength{\cftbeforetoctitleskip}{0pt}
\setlength{\cftaftertoctitleskip}{16pt}

\renewcommand{\cftloftitlefont}{\hfil\fontsize{16pt}{18pt}\selectfont\bfseries}
\renewcommand{\cftafterloftitle}{\hfil}
\setlength{\cftbeforeloftitleskip}{0pt}
\setlength{\cftafterloftitleskip}{16pt}

\renewcommand{\cftlottitlefont}{\hfil\fontsize{16pt}{18pt}\selectfont\bfseries}
\renewcommand{\cftafterlottitle}{\hfil}
\setlength{\cftbeforelottitleskip}{0pt}
\setlength{\cftafterlottitleskip}{16pt}

% Font 13pt tường minh cho tất cả entry trong TOC
\renewcommand{\cftchapfont}{\fontsize{13pt}{15.6pt}\selectfont\bfseries}
\renewcommand{\cftchappagefont}{\fontsize{13pt}{15.6pt}\selectfont\bfseries}
\renewcommand{\cftsecfont}{\fontsize{13pt}{15.6pt}\selectfont}
\renewcommand{\cftsecpagefont}{\fontsize{13pt}{15.6pt}\selectfont}
\renewcommand{\cftsubsecfont}{\fontsize{13pt}{15.6pt}\selectfont}
\renewcommand{\cftsubsecpagefont}{\fontsize{13pt}{15.6pt}\selectfont}

% Chấm dẫn cho chapter
\renewcommand{\cftchapleader}{\cftdotfill{\cftdotsep}}

% Canh lề (indent) trong TOC
\setlength{\cftchapindent}{0pt}       % Chapter: sát lề trái
\setlength{\cftchapnumwidth}{0pt}     % Không dành chỗ số chương (dùng \addcontentsline)
\setlength{\cftsecindent}{1.5em}      % Section: thụt 1.5em
\setlength{\cftsecnumwidth}{2.5em}    % Chỗ số section (1.1)
\setlength{\cftsubsecindent}{4.0em}   % Subsection: thụt 4em
\setlength{\cftsubsecnumwidth}{3.0em} % Chỗ số subsection (1.1.1)

% Font 13pt cho danh mục hình và bảng
\renewcommand{\cftfigfont}{\fontsize{13pt}{15.6pt}\selectfont}
\renewcommand{\cftfigpagefont}{\fontsize{13pt}{15.6pt}\selectfont}
\renewcommand{\cfttabfont}{\fontsize{13pt}{15.6pt}\selectfont}
\renewcommand{\cfttabpagefont}{\fontsize{13pt}{15.6pt}\selectfont}

% Khoảng cách giữa các mục
\setlength{\cftbeforechapskip}{6pt}
\setlength{\cftbeforesecskip}{2pt}
\setlength{\cftbeforesubsecskip}{1pt}

% ===== FORMAT CHAPTER / SECTION =====
% [display]: label (CHƯƠNG X) trên 1 dòng, title trên dòng tiếp
% \chapter*{...} tự động bỏ label → không ảnh hưởng Lời cảm ơn, Kết luận, Phụ lục
\titleformat{\chapter}[display]
  {\centering\fontsize{16pt}{18pt}\bfseries}
  {\chaptername~\thechapter}{6pt}{}
\titlespacing*{\chapter}{0pt}{0pt}{1cm}

\titleformat{\section}
  {\fontsize{13pt}{15.6pt}\bfseries}
  {\thesection}{1em}{}

\titleformat{\subsection}
  {\fontsize{13pt}{15.6pt}\bfseries}
  {\thesubsection}{1em}{}
